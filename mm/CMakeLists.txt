cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vantage_memory_tests LANGUAGES C)
  include(CTest)
endif()

if(BUILD_TESTING)
  add_library(mm_test_core STATIC ranges.c pages.c heap.c fdt.c boot.c)
  target_compile_definitions(mm_test_core PUBLIC VANTAGE_SIM)
  foreach(test_name pages allocators fdt)
    add_executable(mm_${test_name}_test "../sim/tests/mm_${test_name}.c")
    target_link_libraries(mm_${test_name}_test PRIVATE mm_test_core)
    add_test(NAME mm_${test_name} COMMAND mm_${test_name}_test)
  endforeach()
  foreach(target mm_test_core mm_pages_test mm_allocators_test mm_fdt_test)
    set_target_properties(${target} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
  endforeach()
endif()
