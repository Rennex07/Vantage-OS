cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vantage_scheduler_tests LANGUAGES C)
  include(CTest)
endif()

if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(task_smp_test ../../sim/tests/task_smp.c task.c self_test.c ../parallel.c)
  target_compile_definitions(task_smp_test PRIVATE VANTAGE_SIM)
  target_link_libraries(task_smp_test PRIVATE Threads::Threads)
  set_target_properties(task_smp_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
  if(MSVC)
    target_compile_options(task_smp_test PRIVATE /W4 /WX)
  else()
    target_compile_options(task_smp_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME scheduler_four_core COMMAND task_smp_test)
  set_tests_properties(scheduler_four_core PROPERTIES TIMEOUT 120)
endif()
