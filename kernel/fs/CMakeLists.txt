cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vantage_file_tests LANGUAGES C)
  include(CTest)
endif()

if(BUILD_TESTING)
  add_executable(files_test "../../sim/tests/files.c" files.c file_syscall.c storage.c)
  target_compile_definitions(files_test PRIVATE VANTAGE_SIM)
  set_target_properties(files_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
  if(MSVC)
    target_compile_options(files_test PRIVATE /W4 /WX)
  else()
    target_compile_options(files_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME persistent_files COMMAND files_test)
  add_executable(storage_test "../../sim/tests/storage.c" storage.c)
  target_compile_definitions(storage_test PRIVATE VANTAGE_SIM)
  set_target_properties(storage_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
  if(MSVC)
    target_compile_options(storage_test PRIVATE /W4 /WX)
  else()
    target_compile_options(storage_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME storage_recovery COMMAND storage_test)
  add_executable(storage_host_test "../../sim/tests/storage_host.c" "../../sim/platform/storage_host.c" files.c storage.c)
  target_compile_definitions(storage_host_test PRIVATE VANTAGE_SIM)
  set_target_properties(storage_host_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
  if(MSVC)
    target_compile_definitions(storage_host_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(storage_host_test PRIVATE /W4 /WX)
  else()
    target_compile_options(storage_host_test PRIVATE -Wall -Wextra -Werror)
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME storage_process_reboot COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/tools/storage_host_roundtrip.py" "$<TARGET_FILE:storage_host_test>")
  set_tests_properties(persistent_files storage_recovery storage_process_reboot PROPERTIES TIMEOUT 120)
endif()
