cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vantage_driver_tests LANGUAGES C)
  include(CTest)
endif()

if(BUILD_TESTING)
  add_executable(hardware_test ../sim/tests/drivers/hardware.c hw/wait.c hw/property.c)
  add_executable(v3d_test ../sim/tests/drivers/v3d.c video/v3d.c hw/wait.c)
  add_executable(wifi_test ../sim/tests/drivers/wifi.c wifi/cyw43455.c wifi/nvram.c)
  add_executable(sdio_test ../sim/tests/drivers/sdio.c wifi/sdio.c hw/wait.c)
  add_executable(sd_controller_test ../sim/tests/sd_controller.c sd.c)
  add_executable(delays_test ../sim/tests/delays.c ../kernel/sched/delays/delay.c)
  foreach(target hardware_test v3d_test wifi_test sdio_test sd_controller_test delays_test)
    target_compile_definitions(${target} PRIVATE VANTAGE_SIM)
    set_target_properties(${target} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
  endforeach()
  add_test(NAME hardware_mailbox COMMAND hardware_test)
  add_test(NAME v3d_commands_and_mmu COMMAND v3d_test)
  add_test(NAME sdio_host COMMAND sdio_test)
  add_test(NAME wifi_firmware_and_frames COMMAND wifi_test)
  add_test(NAME sd_controller COMMAND sd_controller_test)
  add_test(NAME delays COMMAND delays_test)
  set_tests_properties(hardware_mailbox v3d_commands_and_mmu sdio_host wifi_firmware_and_frames sd_controller delays
    PROPERTIES TIMEOUT 30)
endif()
